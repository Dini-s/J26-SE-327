			String stciv,String prof,double stat,String cap,String eyes,String sp,Date dr,Date ds,boolean validEsp){
		/** the unique code of the identity card requested by the citizen; it is entered by the employee when the paper document is created and when the information has to be kept in the database. */
		this.Numero=cod_carta;
		/** I assign the identity card that I am creating to the citizen who requested it and who is present in the municipal database */
		
		this.Cittadinanza=citt;
		this.Residenza=res;
		this.Via=via;
		this.num_civico=num;
		this.StatoCivile=stciv;
		this.Professione=prof;
		this.Statura=stat;
		this.Capelli=cap;
		this.Occhi=eyes;
		this.SegniParticolari=sp;
		this.DataRilascio=dr;
		this.DataScadenza=ds;
		this.ValidaEspatrio=validEsp;
		
	}

		int month2=0;
