package DB;
import Bean.*;
/** The class DbException is thrown when an exception related to the db occurs @author Antonio Leone @version 1.0 */
public class DbException extends RuntimeException {

	private static final long serialVersionUID = -6403170047487930045L;
	public DbException()
	{
	}
	public DbException(String message)
	{
		super(message);
	}
}
