package smos.exception;

import java.lang.Exception;

/** This class represents the exception raised when an entity is not found in the database. */
public class EntityNotFoundException extends Exception {
	
	private static final long serialVersionUID = -1236105333523133721L;

	/** Raises the exception without an associated error message. */
	public EntityNotFoundException() {
		super("Object Not Found in Repository!");
	}
	
	/** Raises the exception with an associated error message. */
	public EntityNotFoundException(String pMessage) {
		super(pMessage);
	}
	
	
}
