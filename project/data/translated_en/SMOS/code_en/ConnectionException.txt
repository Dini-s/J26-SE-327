package smos.exception;

import java.lang.Exception;

/** This class represents the exception raised when it is not possible to obtain a connection to the database */
public class ConnectionException extends Exception {
	
	private static final long serialVersionUID = -6593436034986073011L;

	/** Raises the exception without an associated error message. */
	public ConnectionException() {
		super("Unable to Connect to the DataBase!");
	}
	
	/** Raises the exception with an associated error message. */
	public ConnectionException(String pMessage) {
		super(pMessage);
	}
	
	
}
